-- 1. HUMANOID CLONE DESYNC, ANIMATIONS & ANTI-RAGDOLL
-----------------------------------
local activeHumanoid = nil

local function applyAntiRagdoll(character, humanoid)
	if not character or not humanoid then return end
	
	humanoid:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
	humanoid:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
	
	for _, descendant in pairs(character:GetDescendants()) do
		if descendant:IsA("Motor6D") then
			descendant.Enabled = true
		end
	end
end

local function applyZombieDesync(character)
	if not character then return end
	savedWalkSpeed = nil
	
	local originalHumanoid = character:WaitForChild("Humanoid", 5)
	if not originalHumanoid then return end
	
	local healthScript = character:FindFirstChild("Health")
	if healthScript then healthScript:Destroy() end
	
	originalHumanoid.BreakJointsOnDeath = false
	
	local clonedHumanoid = originalHumanoid:Clone()
	clonedHumanoid.Name = "Humanoid"
	clonedHumanoid:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
	clonedHumanoid.Health = originalHumanoid.MaxHealth
	
	local animator = clonedHumanoid:FindFirstChildOfClass("Animator")
	if not animator then
		animator = Instance.new("Animator")
		animator.Parent = clonedHumanoid
	end
	
	if Settings.JumpAnimationId and Settings.JumpAnimationId ~= "" then
		pcall(function()
			local jumpAnim = Instance.new("Animation")
			jumpAnim.AnimationId = Settings.JumpAnimationId
			local jumpTrack = animator:LoadAnimation(jumpAnim)

			clonedHumanoid.Jumping:Connect(function(isActive)
				if isActive then
					jumpTrack:Play()
				end
			end)
		end)
	end
	
	originalHumanoid:Destroy()
	clonedHumanoid.Parent = character
	activeHumanoid = clonedHumanoid
	
	applyAntiRagdoll(character, clonedHumanoid)
	
	local camera = workspace.CurrentCamera
	if camera then camera.CameraSubject = clonedHumanoid end
	
	local animateScript = character:FindFirstChild("Animate")
	if animateScript then
		local animateClone = animateScript:Clone()
		animateScript:Destroy()
		animateClone.Parent = character
		animateClone.Disabled = false
	end
end

-- OPTIMIZED: Throttle Heartbeat check to run every ~0.1s to heavily decrease FPS drops
local lastHbCheck = 0
RunService.Heartbeat:Connect(function(dt)
	lastHbCheck = lastHbCheck + dt
	if lastHbCheck < 0.1 then return end
	lastHbCheck = 0

	local char = player.Character
	if char and activeHumanoid then
		for _, descendant in pairs(char:GetDescendants()) do
			if descendant:IsA("Motor6D") and not descendant.Enabled then
				descendant.Enabled = true
			end
		end
		
		local state = activeHumanoid:GetState()
		if state == Enum.HumanoidStateType.Ragdoll or state == Enum.HumanoidStateType.FallingDown then
			activeHumanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
		end
	end
end)

if player.Character then task.spawn(function() applyZombieDesync(player.Character) end) end

local function triggerIncubatorPrompt()
	pcall(function()
		local prompt = workspace:FindFirstChild("IncubatorBloomed")
			and workspace.IncubatorBloomed:FindFirstChild("Root")
			and workspace.IncubatorBloomed.Root:FindFirstChild("PromptAttachment")
			and workspace.IncubatorBloomed.Root.PromptAttachment:FindFirstChildOfClass("ProximityPrompt")

		if prompt then
			prompt.HoldDuration = 0
			prompt.RequiresLineOfSight = false
			if fireproximityprompt then
				fireproximityprompt(prompt)
			end
		end
	end)
end

local function unequipTreadmill()
	pcall(function()
		local network = ReplicatedStorage:FindFirstChild("Network")
		local unequipRemote = network and network:FindFirstChild("Treadmills: RequestUnequip")
		if unequipRemote then
			unequipRemote:InvokeServer()
		end
	end)
end

local function dropEggs()
	task.spawn(function()
		task.wait(Settings.DropEggDelay)
		pcall(function()
			local network = ReplicatedStorage:FindFirstChild("Network")
			if network then
				local dropRemote = network:FindFirstChild("Eggs: RequestAreaEggDrop")
				if dropRemote then
					dropRemote:InvokeServer({ Reason = "PlayerRequest" })
				end
			end
		end)
	end)
end

local function equipBat()
	if not Settings.TreeFarm.AutoEquipBat then return end
	
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not character or not humanoid then return end

	local currentTool = character:FindFirstChildWhichIsA("Tool")
	if currentTool and currentTool.Name:lower():find(Settings.TreeFarm.BatNamePartial:lower()) then
		return
	end

	local backpack = player:FindFirstChild("Backpack")
	if backpack then
		for _, tool in ipairs(backpack:GetChildren()) do
			if tool:IsA("Tool") and tool.Name:lower():find(Settings.TreeFarm.BatNamePartial:lower()) then
				humanoid:EquipTool(tool)
				break
			end
		end
	end
end

-----------------------------------
-- ALL DEBRIS TOUCHINTEREST REMOVER
-----------------------------------
task.spawn(function()
	local workspace = game:GetService("Workspace")

	local function clearAllDebrisTouchInterests()
		pcall(function()
			local debris = workspace:FindFirstChild("__DEBRIS")
			if not debris then return end

			-- Search through every single object in __DEBRIS
			for _, descendant in ipairs(debris:GetDescendants()) do
				if descendant:IsA("TouchTransmitter") or descendant.Name == "TouchInterest" then
					descendant:Destroy()
				end
			end
		end)
	end

	-- Run immediately for existing debris
	clearAllDebrisTouchInterests()

	-- Continuously destroy any new TouchInterests as traps spawn in __DEBRIS
	local debris = workspace:FindFirstChild("__DEBRIS")
	if debris then
		debris.DescendantAdded:Connect(function(descendant)
			if descendant:IsA("TouchTransmitter") or descendant.Name == "TouchInterest" then
				task.wait(0.01)
				descendant:Destroy()
			end
		end)
	end
end)

			if targetObj and targetObj:IsA("TextLabel") and targetObj.Text ~= "" then
				countdownText = targetObj.Text
			end

			if not countdownText then
				local incubator = workspace:FindFirstChild("IncubatorBloomed")
				if incubator then
					for _, descendant in ipairs(incubator:GetDescendants()) do
						if descendant:IsA("TextLabel") and descendant.Text:find("%d+:%d+") then
							countdownText = descendant.Text
							break
						end
					end
				end
			end

			if countdownText then
				local mins, secs = string.match(countdownText, "(%d+):(%d+)")
				if mins and secs then
					timerDisplay.Text = string.format("🌸 in %dm %ds", tonumber(mins), tonumber(secs))
				else
					timerDisplay.Text = "🌸 " .. countdownText
				end
			else
				timerDisplay.Text = "🌸 Searching..."
			end
		end)
		task.wait(1)
	end
end)

local function isPlayerInGroundArea(rootPart)
	local groundArea = workspace:FindFirstChild("__OBJECTS")
		and workspace.__OBJECTS:FindFirstChild("Areas")
		and workspace.__OBJECTS.Areas:FindFirstChild("Ground")

	if not groundArea then return false end

	if groundArea:IsA("Model") then
		local cf, size = groundArea:GetBoundingBox()
		local localPos = cf:PointToObjectSpace(rootPart.Position)
		return math.abs(localPos.X) <= (size.X / 2)
			and math.abs(localPos.Y) <= (size.Y / 2 + 20)
			and math.abs(localPos.Z) <= (size.Z / 2)
	elseif groundArea:IsA("BasePart") then
		local localPos = groundArea.CFrame:PointToObjectSpace(rootPart.Position)
		return math.abs(localPos.X) <= (groundArea.Size.X / 2)
			and math.abs(localPos.Y) <= (groundArea.Size.Y / 2 + 20)
			and math.abs(localPos.Z) <= (groundArea.Size.Z / 2)
	end

	return false
end

-- Checks if player is standing on workspace.__OBJECTS.Build.CherryBlossomZone["1"].BOTTOMS.Part
local function isPlayerOnCherryBlossomBottom(rootPart)
	local cherryPart = workspace:FindFirstChild("__OBJECTS")
		and workspace.__OBJECTS:FindFirstChild("Build")
		and workspace.__OBJECTS.Build:FindFirstChild("CherryBlossomZone")
		and workspace.__OBJECTS.Build.CherryBlossomZone:FindFirstChild("1")
		and workspace.__OBJECTS.Build.CherryBlossomZone["1"]:FindFirstChild("BOTTOMS")
		and workspace.__OBJECTS.Build.CherryBlossomZone["1"].BOTTOMS:FindFirstChild("Part")

	if not cherryPart or not cherryPart:IsA("BasePart") then return false end

	local localPos = cherryPart.CFrame:PointToObjectSpace(rootPart.Position)
	return math.abs(localPos.X) <= (cherryPart.Size.X / 2)
		and math.abs(localPos.Y) <= (cherryPart.Size.Y / 2 + 20)
		and math.abs(localPos.Z) <= (cherryPart.Size.Z / 2)
end

local function restoreWalkSpeed()
	local character = player.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if humanoid and savedWalkSpeed then
		humanoid.WalkSpeed = savedWalkSpeed
		savedWalkSpeed = nil
	end
end

local function stopCurrentTween(shouldDropEgg)
	if activeTweenConn then activeTweenConn:Disconnect() activeTweenConn = nil end
	if activeTween then activeTween:Cancel() activeTween = nil end
	
	local character = player.Character
	local rootPart = character and character:FindFirstChild("HumanoidRootPart")
	if rootPart then rootPart.AssemblyLinearVelocity = Vector3.zero end
	
	restoreWalkSpeed()
	if shouldDropEgg then dropEggs() end
end

local function tweenToTarget(targetPos)
	local character = player.Character
	local rootPart = character and character:FindFirstChild("HumanoidRootPart")
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")

	if rootPart then
		if isPlayerInGroundArea(rootPart) then
			unequipTreadmill()
		end

		local speed = tonumber(speedInput.Text) or Settings.Speed
		stopCurrentTween(false)

		if humanoid then
			if humanoid.WalkSpeed > 0 then savedWalkSpeed = humanoid.WalkSpeed end
			humanoid.WalkSpeed = 0
		end

		local function doTweenStep(destination, onComplete)
			local dist = (destination - rootPart.Position).Magnitude
			if dist < 2 then
				if onComplete then onComplete() end
				return
			end

			local tweenInfo = TweenInfo.new(dist / speed, Enum.EasingStyle.Linear)
			local tween = TweenService:Create(rootPart, tweenInfo, {CFrame = CFrame.new(destination)})
			activeTween = tween

			activeTweenConn = tween.Completed:Connect(function(state)
				if activeTween == tween then
					activeTween = nil
					if activeTweenConn then activeTweenConn:Disconnect() activeTweenConn = nil end
					if state == Enum.PlaybackState.Completed then
						rootPart.AssemblyLinearVelocity = Vector3.zero
						character:PivotTo(CFrame.new(destination))
						if onComplete then onComplete() end
					end
				end
			end)

			tween:Play()
		end

		local waypoint = Settings.Waypoint
		local targetIsWaypoint = (targetPos - waypoint).Magnitude < 2

		if Settings.UseGroundDetection and isPlayerInGroundArea(rootPart) and not targetIsWaypoint then
			doTweenStep(waypoint, function()
				doTweenStep(targetPos, function()
					restoreWalkSpeed()
				end)
			end)
		else
			doTweenStep(targetPos, function()
				restoreWalkSpeed()
			end)
		end
	end
end

local function findNearestTree(rootPart)
	local nearestPrompt, nearestPart, nearestObject = nil, nil, nil
	local shortestDistance = math.huge

	local sakuraFolder = workspace:FindFirstChild(Settings.TreeFarm.SakuraFolderName)
	if sakuraFolder then
		for _, tree in ipairs(sakuraFolder:GetChildren()) do
			local prompt = tree:FindFirstChildWhichIsA("ProximityPrompt", true)
			local part = tree:IsA("BasePart") and tree or tree:FindFirstChildWhichIsA("BasePart", true)
			if part then
				local dist = (part.Position - rootPart.Position).Magnitude
				if dist < shortestDistance then
					shortestDistance = dist
					nearestPrompt, nearestPart, nearestObject = prompt, part, tree
				end
			end
		end
	end

	if nearestPart then return nearestPrompt, nearestPart, nearestObject end

	for _, obj in ipairs(workspace:GetChildren()) do
		local nameLower = obj.Name:lower()
		if nameLower:find("tree") or nameLower:find("wood") or nameLower:find("plant") then
			local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
			local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
			if part then
				local dist = (part.Position - rootPart.Position).Magnitude
				if dist < shortestDistance then
					shortestDistance = dist
					nearestPrompt, nearestPart, nearestObject = prompt, part, obj
				end
			end
		end
	end

	return nearestPrompt, nearestPart, nearestObject
end

local function startTreeFarmLoop()
	task.spawn(function()
		while autoTreeState do
			local character = player.Character
			local rootPart = character and character:FindFirstChild("HumanoidRootPart")

			if rootPart then
				local nearestPrompt, nearestPart, nearestObject = findNearestTree(rootPart)

				if nearestPart and nearestObject then
					equipBat()

					-- Route to intermediate coordinate only ONCE if not already on the platform
					if not isPlayerOnCherryBlossomBottom(rootPart) and Settings.TreeFarm.IntermediateTP then
						tweenToTarget(Settings.TreeFarm.IntermediateTP)
						while activeTween do task.wait(0.1) end
					end

					-- Directly tween to the targeted tree
					if autoTreeState and nearestObject and nearestObject.Parent then
						local targetPos = nearestPart.Position + Vector3.new(0, Settings.TreeFarm.YOffset, 0)
						tweenToTarget(targetPos)

						while autoTreeState and nearestObject and nearestObject.Parent do
							local currentCharacter = player.Character
							local currentRoot = currentCharacter and currentCharacter:FindFirstChild("HumanoidRootPart")

							if currentRoot and nearestPart and nearestPart.Parent then
								equipBat()
								local distToTree = (nearestPart.Position - currentRoot.Position).Magnitude
								if distToTree > Settings.TreeFarm.ReTweenDistance and not activeTween then
									tweenToTarget(nearestPart.Position + Vector3.new(0, Settings.TreeFarm.YOffset, 0))
								end
							end

							if nearestPrompt and nearestPrompt.Parent and fireproximityprompt then
								fireproximityprompt(nearestPrompt)
							end

							task.wait(Settings.TreeFarm.InteractInterval)
						end
					end

					if activeTween then stopCurrentTween(false) end
				else
					task.wait(0.5)
				end
			else
				task.wait(0.5)
			end
		end
	end)
end

if autoTreeState then startTreeFarmLoop() end

-- Function to make a proximity prompt instant
local function makePromptInstant(prompt)
    if prompt:IsA("ProximityPrompt") then
        -- Set hold duration to 0 (instant activation)
        prompt.HoldDuration = 0
        
        -- Make it easier to activate
        prompt.MaxActivationDistance = math.max(prompt.MaxActivationDistance, 10)
        
        -- Remove any requirement for line of sight
        prompt.RequiresLineOfSight = false
    end
end

-- Make all existing prompts instant
for _, descendant in pairs(workspace:GetDescendants()) do
    makePromptInstant(descendant)
end

-- Make new prompts instant when they're added
workspace.DescendantAdded:Connect(function(descendant)
    if descendant:IsA("ProximityPrompt") then
        task.wait(0.05)
        makePromptInstant(descendant)
    end
end)

-- Monitor if prompts get reset and re-apply
workspace.DescendantAdded:Connect(function(descendant)
    if descendant:IsA("ProximityPrompt") then
        descendant:GetPropertyChangedSignal("HoldDuration"):Connect(function()
            if descendant.HoldDuration > 0 then
                descendant.HoldDuration = 0
            end
        end)
    end
end)
